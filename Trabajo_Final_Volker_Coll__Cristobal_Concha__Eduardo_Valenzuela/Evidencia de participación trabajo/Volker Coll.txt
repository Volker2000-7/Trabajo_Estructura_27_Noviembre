class Nodo:
    def init(self)-> None:
        self.inicio=None
class ListaEnlazada:
    def init(self) -> None:
        self.inicio = None

    def insertar_final(self, dato) -> None:
        nuevo = Nodo(dato)
        if self.inicio is None:
            self.inicio = nuevo
        else:
            puntero = self.inicio
            while puntero.siguiente is not None:
                puntero = puntero.siguiente
            puntero.siguiente = nuevo

"""Funciones buses"""
def cargar_buses(nombre_archivo: str, lista_buses) -> None:
    try:
        # Recorre toda la lista enlazada de buses  y la añade al archivo CSV
        archivoBuses = open(nombre_archivo, "r", encoding=ENCODE) # Abre el archivo de buses
        lineas = archivoBuses.readlines()   # Guarda las lineas del archivo en la variable lineas
        archivoBuses.close() # Se cierra
        for linea in lineas[1:]: # Se salta la primera linea
            linea_limpia = linea.strip() # Limpia los espacios izq-der de las lineas
            if linea_limpia != "": # Valida que no este vacia
                datos = linea_limpia.split(",") #Separa los datos por comas
                id_bus = int(datos[0]) 
                patente = datos[1]
                capacidad = int(datos[2])
                if datos[3] == "True": # Si el estado es True se pasa a booleano
                    estado = True
                else:
                    estado = False
                nuevo_bus = buses(id_bus, patente, capacidad, estado) # Se guarda todo en la clase buses
                lista_buses.insertar_final(nuevo_bus) # Se inserta el nuevo bus en la lista enlazada de buses
    except FileNotFoundError: # Si el archivo no existe cae en el except y lo crea
        archivo = open(nombre_archivo, "w", encoding=ENCODE)
        archivo.write("idBus,patente,capacidad,disponibilidad\n")
        archivo.close()       
def registrar_bus(lista_buses: ListaEnlazada, nombre_archivo: str, id_bus: int, patente: str, capacidad: int, disponibilidad: bool) -> None:
    try:
        if id_bus <= 0:
            raise ValueError("El ID debe ser mayor a 0.") #valida que sea positivo
        puntero = lista_buses.inicio
        while puntero is not None: # recorremos la lista enlazada para ver si el id_bus esta repetido
            if puntero.dato.idBus == id_bus:
                raise ValueError("Ya existe un bus con ese ID.")
            puntero = puntero.siguiente
        if patente == "":
            raise ValueError("La patente no puede estar vacía.") # valida que la patente no entre vacia
        puntero = lista_buses.inicio
        while puntero is not None:
            if puntero.dato.patente == patente:
                raise ValueError("Ya existe un bus con esa patente.")  # recorre la lista enlazada para asegurar que no exista otro bus con la misma patente
            puntero = puntero.siguiente
        if capacidad <= 0:
            raise ValueError("La capacidad debe ser un número entero positivo.") #valida que la capacidad sea mayor a cero
        # creamos y guardamos el bus
        nuevo_bus = buses(id_bus, patente, capacidad, disponibilidad)
        lista_buses.insertar_final(nuevo_bus)
        guardar_buses(nombre_archivo, lista_buses)
        print("Bus registrado")
    except ValueError as err:
        print(f"Error: {err}")  

def guardar_buses(nombre_archivo: str, lista_buses: ListaEnlazada) -> None:
    # se abre con w para actualizarlo con la lista enlazada
    archivoBuses = open(nombre_archivo, "w", encoding=ENCODE)
    # escribimos la primera fila con los nombres exactos de las columnas
    archivoBuses.write("idBus,patente,capacidad,disponibilidad\n")
    # recorremos la lista enlazada 
    puntero = lista_buses.inicio
    while puntero is not None:
        bus = puntero.dato  # bus es el objeto 'buses' guardado en el nodo actual
        # escribe los 4 atributos separados por comas y con salto de linea
        archivoBuses.write(f"{bus.idBus},{bus.patente},{bus.capacidad},{bus.disponibilidad}\n")
        # avanzamos al siguiente nodo de la lista
        puntero = puntero.siguiente
    # cerramos el archivo 
    archivoBuses.close()

"""Funciones de verificacion de datos ingresados"""
def verificar_palabras(palabra:str)->bool:
    limpiezita=palabra.strip()# le quito los espacios a la palabra
    if len(limpiezita)>0 and limpiezita.replace(" ","").isalpha():#el largo de la palabra debe ser mayor a 0 y el replace le quita los espacios y la funcion isalpha() verifica qye sea realmente una palabra y no un numero
        return True 
    else: 
        return False 
def verificar_numeros(numero:str)->bool:
    limpiezito=numero.strip()#le quito los espacios en blanco 
    if int(limpiezito)>0 and limpiezito.isdigit():#convierto el numero de str a int y verifico que sea mayor a 0 y la funcion isdigit() verifica si el numero sea realmente un numero y no una palabra
        return True 
    else:
        return False

#FUNCION AGREGAR LOS NODOS A LAS LISTAS
    def agregar(self,nuevo):
        if self.inicio==None:
            self.inicio=nuevo
        else:
            actual=self.inicio
            while actual.siguiente!=None:
                actual=actual.siguiente
            actual.siguiente=nuevo

        if idBus <= 0:
            raise ValueError("El ID tiene que ser mayor a 0")

        if capacidad <= 0:

def anularPasaje(Pasajes: ListaEnlazada, id_pasaje: int):
    try:
        actual = Pasajes.inicio
        while actual is not None:
            if actual.id_pasaje == id_pasaje:
                if actual.estado == "Anulado":
                    raise ValueError("El pasaje ya esta anulado.")
                actual.estado = "Anulado"
                guardar_pasajes(Pasajes)
                print("Pasaje anulado!!.")
                return
            actual = actual.siguiente
        raise ValueError("No existe ese pasaje.")
    except ValueError as error:
        print(f"Error: {error}")

def reporte_pasajes_y_recaudacion_viaje(Pasajes: ListaEnlazada,Viajes: ListaEnlazada,Clientes: ListaEnlazada,Ciudades: ListaEnlazada) -> None:
    try:
        Viajes.mostrarViajes(Ciudades)
        id_viaje: int = id_menu_1_2("Ingrese ID de viaje a consultar: ")
        viaje: Viaje = Viajes.buscarViaje(id_viaje)
        origen: str = Ciudades.buscarCiudad(viaje.origen).nombre
        destino: str = Ciudades.buscarCiudad(viaje.destino).nombre

        print(Fore.CYAN + Style.BRIGHT + f"\n REPORTE DE PASAJES Y RECAUDACIÓN - VIAJE #{id_viaje} ({origen} -> {destino})")
        actual = Pasajes.inicio
        recaudacion_vigente: int = 0
        cont_vigentes: int = 0
        cont_anulados: int = 0

        while actual is not None:
            if actual.id_viaje == id_viaje:
                cliente: Cliente = Clientes.buscarCliente(actual.id_cliente)
                nom_origen: str = Ciudades.buscarCiudad(actual.id_origen).nombre
                nom_destino: str = Ciudades.buscarCiudad(actual.id_destino).nombre

                if actual.estado == "Vigente":
                    etiqueta: str = Fore.GREEN + "[VIGENTE]"
                    recaudacion_vigente += actual.valor
                    cont_vigentes += 1
                else:
                    etiqueta = Fore.RED + "[ANULADO]"
                    cont_anulados += 1

                print(
                    f"{etiqueta}{Style.RESET_ALL} Pasaje #{actual.id_pasaje} | "
                    f"Asiento: {actual.asiento} | "
                    f"Pasajero: {cliente.nombre} {cliente.apellido_paterno} | "
                    f"Tramo: {nom_origen} -> {nom_destino} | "
                    f"Valor: ${actual.valor}"
                )
            actual = actual.siguiente

        if cont_vigentes == 0 and cont_anulados == 0:
            print(Fore.YELLOW + "Viaje aún no registra pasajes vendidos ni anulados.")
        else:
            print(Fore.CYAN + "-------------------------------------------------------------------------")
            print(f"Pasajes Vigentes: {Fore.GREEN}{cont_vigentes}{Style.RESET_ALL} | Pasajes Anulados: {Fore.RED}{cont_anulados}")
            print(Fore.GREEN + Style.BRIGHT + f"RECAUDACIÓN VIGENTE DEL VIAJE: ${recaudacion_vigente}")
            print(Fore.CYAN + "-------------------------------------------------------------------------")
    except Exception as error:
        print(Fore.RED + f"Error al generar reporte del viaje: {error}")
